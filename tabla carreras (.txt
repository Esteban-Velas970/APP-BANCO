CREATE TABLE carreras (
    id INT AUTO_INCREMENT NOT NULL,
    nombre VARCHAR(50) NOT NULL,
    PRIMARY KEY (id)
);

CREATE TABLE estudiantes (
    id INT AUTO_INCREMENT NOT NULL,
    nombre VARCHAR(50) NOT NULL,
    carrera_id INT,
    PRIMARY KEY (id),
    FOREIGN KEY (carrera_id) REFERENCES carreras(Id)
);
